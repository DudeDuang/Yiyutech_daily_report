```python
def build_prompt(manifest, record):
    key = f"{record['ageMonths']}:{record['dimension']}"
    storyboard = manifest["storyboardTemplates"][key]
    shots = "\n".join(
        f"{shot['time']}｜{shot['shot']}｜{shot['action']}"
        for shot in storyboard["shots"]
    )
    return f"{record['seedancePrompt']}\n【时间轴】\n{shots}"
```
